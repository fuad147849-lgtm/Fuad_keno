import { DurableObject } from "cloudflare:workers"; export class MyDurableObject extends DurableObject { constructor(ctx, env) { super(ctx, env); } async fetch(request) { const url = new URL(request.url); if (url.pathname === "/state") { this.ctx.storage.sql.exec(` CREATE TABLE IF NOT EXISTS game_state ( id TEXT PRIMARY KEY, value TEXT NOT NULL ) `); let row = this.ctx.storage.sql .exec("SELECT value FROM game_state WHERE id = ?", "main") .one(); let state; if (row) { state = JSON.parse(row.value); } else { state = { round: 1, status: "waiting", players: {}, tickets: {}, drawnNumbers: [] }; this.ctx.storage.sql.exec( "INSERT INTO game_state (id, value) VALUES (?, ?)", "main", JSON.stringify(state) ); } return Response.json(state); } return new Response("FUADX KENO SERVER READY"); } } export default { async fetch(request, env) { const id = env.MY_DURABLE_OBJECT.idFromName("fuadx-keno-game"); const stub = env.MY_DURABLE_OBJECT.get(id); return stub.fetch(request); } };
